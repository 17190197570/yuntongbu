cmake_minimum_required(VERSION 3.24)
project(CloudProviderHost LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "CloudProviderHost requires Windows and the Windows SDK.")
endif()
add_executable(CloudProviderHost main.cpp CloudProviderHost.manifest)
target_compile_features(CloudProviderHost PRIVATE cxx_std_17)
target_compile_definitions(CloudProviderHost PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 NTDDI_VERSION=0x0A000004)
target_link_libraries(CloudProviderHost PRIVATE CldApi Bcrypt)
if(MSVC)
  target_compile_options(CloudProviderHost PRIVATE /W4 /permissive- /EHsc /utf-8)
  set_property(TARGET CloudProviderHost PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
if(MINGW)
  target_link_options(CloudProviderHost PRIVATE -municode)
endif()
